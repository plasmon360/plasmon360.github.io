% Loads a data for the example 
load mandrill

% Plots an image for the example 
image(X)

% Inserts a colorbar. a handle is created 
t=colorbar;

% sets the ylabel property of the handle t. 
set(get(t,'ylabel'),'string','My colorbar title','Fontsize',10)
